```ABAP
METHOD CREATE_BASIC_OBJECTS.
  DATA  L_APP_NAME(200).
  DATA: LO_BDS_INSTANCE      TYPE REF TO CL_BDS_DOCUMENT_SET.
  DATA: LDOC_SIGNATURE    TYPE SBDST_SIGNATURE,
        LWA_DOC_SIGNATURE LIKE LINE OF LDOC_SIGNATURE,
        LS_DOC_COMPONENTS TYPE SBDST_COMPONENTS,
        LT_DOC_URIS       TYPE SBDST_URI,
        LWA_DOC_URIS      LIKE LINE OF LT_DOC_URIS.

  DATA: LHAS TYPE I.
  DATA:L_CONTAINER_NAME TYPE CHAR20 VALUE 'CONTAINER'.

  CLEAR E_ERROR.
* first get the SAP DOI i_oi_container_control interface
  CALL METHOD C_OI_CONTAINER_CONTROL_CREATOR=>GET_CONTAINER_CONTROL
    IMPORTING
      CONTROL = GO_CONTROL
      ERROR   = E_ERROR.

* check no errors occured
  IF E_ERROR IS NOT INITIAL.
    CHECK E_ERROR->HAS_SUCCEEDED EQ 'X'.
  ENDIF.

  IF I_CONTAINER_NAME IS SUPPLIED.
    L_CONTAINER_NAME = I_CONTAINER_NAME.
  ENDIF.

  CREATE OBJECT GO_CONTAINER
    EXPORTING
      CONTAINER_NAME = L_CONTAINER_NAME.

  IF I_APP_NAME IS INITIAL.
    L_APP_NAME = 'Excel报表输出数据'.
  ELSE.
    L_APP_NAME = I_APP_NAME.
  ENDIF.
*& 重构 Begin of modification by ey_chenxh 2026.04.22
*& 如果屏幕里面没有container设置，则默认当前屏幕作为容器
  IF I_CONTAINER_NAME IS NOT INITIAL.
    CALL METHOD GO_CONTROL->INIT_CONTROL
      EXPORTING
        R3_APPLICATION_NAME      = L_APP_NAME
        INPLACE_ENABLED          = 'X'
        INPLACE_SCROLL_DOCUMENTS = 'X'
        PARENT                   = GO_CONTAINER
        REGISTER_ON_CLOSE_EVENT  = 'X'
        REGISTER_ON_CUSTOM_EVENT = 'X'
        NO_FLUSH                 = 'X'
      IMPORTING
        ERROR                    = E_ERROR.
  ELSE.
    CALL METHOD GO_CONTROL->INIT_CONTROL
      EXPORTING
        R3_APPLICATION_NAME      = L_APP_NAME
        INPLACE_ENABLED          = 'X'
        INPLACE_SCROLL_DOCUMENTS = 'X'
        PARENT                   = CL_GUI_CUSTOM_CONTAINER=>DEFAULT_SCREEN
        REGISTER_ON_CLOSE_EVENT  = 'X'
        REGISTER_ON_CUSTOM_EVENT = 'X'
        NO_FLUSH                 = 'X'
      IMPORTING
        ERROR                    = E_ERROR.
  ENDIF.
*& 重构 End of modification by ey_chenxh 2026.04.22
* check no errors occured
  IF E_ERROR IS NOT INITIAL.
    CHECK E_ERROR->HAS_SUCCEEDED EQ 'X'.
  ENDIF.

*  gdocument_type = cns_excel.
  LWA_DOC_SIGNATURE-PROP_VALUE = I_DOCNAME.

  APPEND LWA_DOC_SIGNATURE TO LDOC_SIGNATURE.
  CREATE OBJECT LO_BDS_INSTANCE.
  CALL METHOD LO_BDS_INSTANCE->GET_INFO
    EXPORTING
      CLASSNAME  = I_CLASSNAME
      CLASSTYPE  = I_CLASSTYPE
      OBJECT_KEY = I_OBJECT_KEY
    CHANGING
      COMPONENTS = LS_DOC_COMPONENTS
      SIGNATURE  = LDOC_SIGNATURE.

  CALL METHOD LO_BDS_INSTANCE->GET_WITH_URL
    EXPORTING
      CLASSNAME  = I_CLASSNAME
      CLASSTYPE  = I_CLASSTYPE
      OBJECT_KEY = I_OBJECT_KEY
    CHANGING
      URIS       = LT_DOC_URIS
      SIGNATURE  = LDOC_SIGNATURE.

  FREE LO_BDS_INSTANCE.
  READ TABLE LT_DOC_URIS INTO LWA_DOC_URIS INDEX 1.

* ask the SAP DOI container for a i_oi_document_proxy for Excel
  CLEAR E_ERROR.
  CALL METHOD GO_CONTROL->GET_DOCUMENT_PROXY
    EXPORTING
      DOCUMENT_TYPE  = 'Excel.Sheet'
      NO_FLUSH       = 'X'
    IMPORTING
      DOCUMENT_PROXY = GO_DOCUMENT
      ERROR          = E_ERROR.
* check no errors occured
  IF E_ERROR IS NOT INITIAL.
    CHECK E_ERROR->HAS_SUCCEEDED EQ 'X'.
  ENDIF.

* open a document saved in business document service.
  CALL METHOD GO_DOCUMENT->OPEN_DOCUMENT
    EXPORTING
      OPEN_INPLACE = 'X'
      DOCUMENT_URL = LWA_DOC_URIS-URI.

  CALL METHOD GO_DOCUMENT->HAS_SPREADSHEET_INTERFACE
    EXPORTING
      NO_FLUSH     = ''
    IMPORTING
      IS_AVAILABLE = LHAS
      ERROR        = E_ERROR.
* check no errors occured
  IF E_ERROR IS NOT INITIAL.
    CHECK E_ERROR->HAS_SUCCEEDED EQ 'X'.
  ENDIF.

  CALL METHOD GO_DOCUMENT->GET_SPREADSHEET_INTERFACE
    EXPORTING
      NO_FLUSH        = ''
    IMPORTING
      SHEET_INTERFACE = GO_SPREADSHEET
      ERROR           = E_ERROR.

ENDMETHOD.
```

```ABAP
METHOD fill_cell.
  DATA: lt_ranges       TYPE soi_range_list.
  DATA: lwa_range     LIKE LINE OF lt_ranges.

  DATA: lt_contents     TYPE soi_generic_table.
  DATA: lwa_content     LIKE LINE OF lt_contents.
  DATA:l_row TYPE i,
       l_col TYPE i.

  CHECK go_spreadsheet IS NOT INITIAL.
*===================================
* 选择sheet
*===================================
  IF i_sheet_name IS SUPPLIED.
    g_selected_sheet_name = i_sheet_name.
  ENDIF.

  me->select_sheet(
    EXPORTING
      i_sheet_name = g_selected_sheet_name
    IMPORTING
      e_error      = e_error
         ).

  IF e_error IS NOT INITIAL.
    CHECK e_error->has_succeeded EQ 'X'.
  ENDIF.

*===================================
* 插入range
*===================================
  MOVE i_row TO l_row.
  MOVE i_col TO l_col.
* 选择区域
  CALL METHOD go_spreadsheet->set_selection
    EXPORTING
*     坐标
      top     = l_row  "行
      left    = l_col  "列
*     行数和列数
      rows    = 1  "行数
      columns = 1. "列数

* 创建Range
  CALL METHOD go_spreadsheet->insert_range
    EXPORTING
      name     = 'CELL'
      rows     = 1
      columns  = 1
      no_flush = ''
    IMPORTING
      error    = e_error.

  IF e_error IS NOT INITIAL.
    CHECK e_error->has_succeeded EQ 'X'.
  ENDIF.

* 区域以及位置坐标
  CLEAR lt_ranges[].
  lwa_range-name = 'CELL'.  "Range名称
  lwa_range-rows = 1.        "行数
  lwa_range-columns = 1.     "列数
  APPEND lwa_range TO lt_ranges.

*===================================
* 填充range
*===================================
  lwa_content-row = i_row.
  lwa_content-column = i_col.
  lwa_content-value = i_value.
  APPEND lwa_content TO lt_contents.

  CALL METHOD go_spreadsheet->set_ranges_data
    EXPORTING
      ranges   = lt_ranges
      contents = lt_contents
    IMPORTING
      error    = e_error.

  CLEAR:lt_ranges[], lt_contents[].
ENDMETHOD.
```

```ABAP
METHOD fill_items.
  DATA: lv_linecnt TYPE i,
        lv_cols TYPE i.
  DATA  l_retcode       TYPE soi_ret_string .
  DATA: lt_ftab        TYPE TABLE OF rfc_fields.

  CHECK go_spreadsheet IS NOT INITIAL.

*===================================
* 选择sheet
*===================================
  IF sheet_name IS SUPPLIED.
    g_selected_sheet_name = sheet_name.
  ENDIF.

  me->select_sheet(
    EXPORTING
      i_sheet_name = g_selected_sheet_name
    IMPORTING
      e_error      = e_error
         ).

  IF e_error IS NOT INITIAL.
    CHECK e_error->has_succeeded EQ 'X'.
  ENDIF.

*===================================
* 得到总行数
*===================================
  IF row_count IS SUPPLIED.
    lv_linecnt = row_count.
  ELSE.
    DESCRIBE TABLE it_out LINES lv_linecnt.
  ENDIF.


  CHECK lv_linecnt GT 0.

*===================================
* 得到it_out的结构
*===================================
  CALL FUNCTION 'DP_GET_FIELDS_FROM_TABLE'
    TABLES
      data   = it_out
      fields = lt_ftab.

  IF col_count IS SUPPLIED.
    lv_cols = col_count.
  ELSE.
    DESCRIBE TABLE lt_ftab LINES lv_cols.
  ENDIF.

*===================================
* 插入range范围
*===================================
  CLEAR e_error.
  CALL METHOD go_spreadsheet->insert_range_dim
    EXPORTING
      name     = 'RANGE1'
      no_flush = 'X'
      top      = begin_row
      left     = begin_col
      rows     = lv_linecnt
      columns  = lv_cols
    IMPORTING
      error    = e_error.

*  IF e_error IS NOT INITIAL.
*    CHECK e_error->has_succeeded EQ 'X'.
*  ENDIF.

*===================================
* 填入数据
*===================================
  CALL METHOD go_spreadsheet->insert_one_table
    EXPORTING
      data_table   = it_out
      fields_table = lt_ftab
      rangename    = 'RANGE1'
      no_flush     = 'X'
      wholetable   = 'X'
    IMPORTING
      error        = e_error
      retcode      = l_retcode.
  .
ENDMETHOD.
```
