# 物料批次特性CDS视图
<!-- more -->

## 1. 物料批次特性CDS视图
### 1.1 物料批次特性CDS视图
比起用BAPI_OBJCL_GETDETAIL获取特性速度快点
```ABAP
@AbapCatalog.sqlViewName: 'ZVMM_MAT_CHARVAL'
@AbapCatalog.compiler.compareFilter: true
@AccessControl.authorizationCheck: #CHECK
@EndUserText.label: '物料批次特性查询'
define view ZVMM_MATCHARVAL
  as select from    inob
    inner join      ausp as a on a.objek = inob.cuobj
    left outer join cabn      on cabn.atinn = a.atinn
{

  cast(LEFT(inob.objek, 40) as matnr)                                         as matnr,
  --工厂
  cast(case inob.klart
    when '023' then ''
    when '022' then SUBSTRING(inob.objek, 41, 4)
    else  ''
  end                                 as werks_d)                             as werks,
  --批次
  cast(case inob.klart
   when '023' then SUBSTRING(inob.objek, 41, 10)
   when '022' then SUBSTRING(inob.objek, 45, 10)
   else  ''
  end                                             as charg_d)                 as charg,
  a.atinn,
  a.atzhl,
  inob.klart,
  inob.obtab,
  a.atwrt,
  a.atflv,
  a.atflb,
  cabn.atnam,
  cabn.atfor,
  cabn.anzst,
  cabn.anzdz
}
where
     inob.klart = '023'
  or inob.klart = '022'
```

```ABAP
@AbapCatalog.sqlViewName: 'ZVMM_BATCHDETIAL'
@AbapCatalog.compiler.compareFilter: true
@AccessControl.authorizationCheck: #CHECK
@EndUserText.label: '物料批次特性查询'
define view ZVMM_BATCH_DETIAL
  as select from  ZVMM_MATCHARVAL as A
    left outer join cabnt on  A.atinn     = cabnt.atinn
                          and cabnt.spras = '1' --$session.system_language
{
  A.matnr,
  A.werks,
  A.charg,
  A.atinn,
  A.atzhl, --
  A.klart,
  A.obtab,
  A.atwrt,
  A.atflv,
  A.atflb,
  A.atnam,
  cabnt.atbez,
  A.atfor,
  A.anzst,
  A.anzdz
}
```
